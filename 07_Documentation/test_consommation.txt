    AVG(f.temperature) as temperature_moyenne

SELECT

### **Étape 3 : Tester quelques requêtes d'analyse (KPI du projet)**

```sql
-- 1. KPI: Consommation totale par région
SELECT 
    r.nom_region,
    SUM(f.consommation_valeur) as consommation_totale,
    SUM(f.cout_energie) as cout_total
FROM FAIT_CONSOMMATION f
JOIN DIM_REGION r ON f.id_region = r.id_region
GROUP BY r.nom_region
ORDER BY consommation_totale DESC;

-- 2. KPI: Consommation par type d'énergie et par bâtiment
SELECT 
    b.nom_batiment,
    e.type_energie,
    SUM(f.consommation_valeur) as consommation_totale
FROM FAIT_CONSOMMATION f
JOIN DIM_BATIMENT b ON f.id_batiment = b.id_batiment
JOIN DIM_ENERGIE e ON f.id_energie = e.id_energie
GROUP BY b.nom_batiment, e.type_energie
ORDER BY b.nom_batiment, e.type_energie;

-- 3. KPI: Évolution dans le temps (par mois)
SELECT 
    t.annee,
    t.mois,
    SUM(f.consommation_valeur) as consommation_mensuelle
FROM FAIT_CONSOMMATION f
JOIN DIM_TEMPS t ON f.id_temps = t.id_temps
GROUP BY t.annee, t.mois
ORDER BY t.annee, t.mois;
```

HAVING COUNT(*) > 1;

SELECT COUNT(*) as lignes_orphelines
FROM FAIT_CONSOMMATION f
LEFT JOIN DIM_TEMPS t ON f.id_temps = t.id_temps
LEFT JOIN DIM_REGION r ON f.id_region = r.id_region
LEFT JOIN DIM_BATIMENT b ON f.id_batiment = b.id_batiment
LEFT JOIN DIM_ENERGIE e ON f.id_energie = e.id_energie
LEFT JOIN DIM_COMPTEUR c ON f.id_compteur = c.id_compteur
